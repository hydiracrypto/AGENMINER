    63→    63→function newPairingCode(): string {
    64→    64→  let c = "";
    65→    65→  for (let i = 0; i < 6; i++) c += Math.floor(Math.random() * 16).toString(16);
    66→    66→  return c;
    67→    67→}

    70→    70→function randomEvmAddress(): string {
    71→    71→  const hex = "0123456789abcdef";
    72→    72→  let a = "0x";
    73→    73→  for (let i = 0; i < 40; i++) a += hex[Math.floor(Math.random() * 16)];
    74→    74→  return a;
    75→    75→}

   182→   182→    setWcUri("");

   259→   259→            clearUriTimer();

   263→   263→            setSdkErr(/reject|denied/i.test(msg) ? ERR_REJECTED.current : ERR_SDK.current);
   264→   264→          });
   265→   265→      }
   266→   266→    } catch {
   267→   267→      if (!binanceCancelled.current && mounted.current) {
   268→   268→        setSdkStatus("failed");
   269→   269→        setSdkErr(t("sdkFailed"));
   270→   270→      }
   271→   271→    }
   272→   272→  }, [clearUriTimer, handleBinanceAccounts, renderQr, t]);
   273→   273→

   289→   289→  /** Terapkan sesi login dari hasil approve (tanpa membuat user ganda) */

   293→   293→        address: user.walletAddress,
   294→   294→        walletName: "Binance Agent Wallet",

   302→   302→
   303→   303→  /** Approve dari perangkat yang sama (tombol Confirm) — lewat API agar sesi sama */
   304→   304→  const approveHere = useCallback(async () => {
   305→   305→    if (!pairCodeRef.current) return;
   306→   306→    setPhase("confirming");
   307→   307→    setError(null);
   308→   308→    try {
   309→   309→      const res = await fetch("/api/pair/approve", {
   310→   310→        method: "POST",
   311→   311→        headers: { "Content-Type": "application/json" },
   312→   312→        body: JSON.stringify({ code: pairCodeRef.current }),
   313→   313→      });
   314→   314→      const d = await res.json();
   315→   315→      if (!res.ok || !d?.ok) {
   316→   316→        if (res.status === 410) {
   317→   317→          setPhase("expired");
   318→   318→        } else {
   319→   319→          setError(d?.error ?? t("errLogin"));
   320→   320→          setPhase("connecting");
   321→   321→        }
   322→   322→        return;
   323→   323→      }
   324→   324→      await new Promise((r) => setTimeout(r, 1200)); // animasi konfirmasi singkat
   325→   325→      applySession(d.user);
   326→   326→    } catch {
   327→   327→      if (mounted.current) {
   328→   328→        setError(t("errLogin"));
   329→   329→        setPhase("connecting");
   330→   330→      }
   331→   331→    }
   332→   332→  }, [applySession, t]);
   333→   333→
   334→   334→  const startPolling = useCallback(() => {
   335→   335→    stopPoll();

   337→   337→      if (!pairCodeRef.current) return;
   338→   338→      try {
   339→   339→        const res = await fetch(`/api/pair/status?code=${pairCodeRef.current}`);
   340→   340→        const d = await res.json();
   341→   341→        if (!mounted.current) return;
   342→   342→        if (d?.status === "APPROVED" && d?.user) {
   343→   343→          stopPoll();
   344→   344→          applySession(d.user);
   345→   345→        } else if (d?.status === "EXPIRED" || d?.status === "NOT_FOUND") {
   346→   346→          stopPoll();
   347→   347→          setPhase("expired");
   348→   348→        }
   349→   349→      } catch {
   350→   350→        /* jaringan sempat gagal → coba di tick berikutnya */
   351→   351→      }

   371→   371→        startPolling();
   372→   372→      })
   373→   373→      .catch(() => {
   374→   374→        startPolling(); // tetap polling; approve tanpa create → NOT_FOUND → expired
   375→   375→      });
   376→   376→  }, [refCode, startPolling]);
   377→   377→
   378→   378→  // ── ganti metode ──
   379→   379→  // Ref ke fungsi terbaru (identitas stabil) agar effect buka-modal

   381→   381→  const latestRef = useRef({ startBinance, regenPair, stopPoll, clearUriTimer });
   382→   382→  useEffect(() => {
   383→   383→    latestRef.current = { startBinance, regenPair, stopPoll, clearUriTimer };
   384→   384→  });
   385→   385→
   386→   386→  const switchMethod = useCallback(
   387→   387→    (m: Method) => {
   388→   388→      if (m === method) return;
   389→   389→      setMethod(m);
   390→   390→      setError(null);
   391→   391→      setQrData("");
   392→   392→      if (m === "pair") {
   393→   393→        binanceCancelled.current = true; // hentikan alur SDK
   394→   394→        latestRef.current.clearUriTimer();
   395→   395→        latestRef.current.regenPair();
   396→   396→      } else {
   397→   397→        latestRef.current.stopPoll();
   398→   398→        latestRef.current.startBinance();
   399→   399→      }
   400→   400→    },
   401→   401→    [method]
   402→   402→  );
   403→   403→
   404→   404→  // Reset tiap kali modal dibuka (HANYA saat `open` berubah)
   405→   405→  useEffect(() => {
   406→   406→    if (!open) return;
   407→   407→    mounted.current = true;
   408→   408→    setShowBrowser(false);
   409→   409→    setOptions([]);
   410→   410→    setCopiedPair(false);
   411→   411→    setMethod("binance");
   412→   412→    latestRef.current.startBinance();
   413→   413→    // deteksi wallet browser secara diam-diam untuk fallback
   414→   414→    getWalletOptions()
   415→   415→      .then((opts) => {
   416→   416→        if (mounted.current) setOptions(opts);
   417→   417→      })
   418→   418→      .catch(() => {});
   419→   419→    return () => {
   420→   420→      mounted.current = false;
   421→   421→      binanceCancelled.current = true;
   422→   422→      latestRef.current.clearUriTimer();
   423→   423→      latestRef.current.stopPoll();
   424→   424→    };
   425→   425→  }, [open]);
   426→   426→
   427→   427→  // Countdown 300s → expired (hanya alur pairing server)
   428→   428→  useEffect(() => {
   429→   429→    if (!open || method !== "pair" || phase !== "connecting") return;
   430→   430→    const iv = setInterval(() => {
   431→   431→      setSecsLeft((s) => {
   432→   432→        if (s <= 1) {
   433→   433→          clearInterval(iv);
   434→   434→          stopPoll();
   435→   435→          if (mounted.current) setPhase("expired");
   436→   436→          return 0;
   437→   437→        }
   438→   438→        return s - 1;
   439→   439→      });
   440→   440→    }, 1000);
   441→   441→    return () => clearInterval(iv);
   442→   442→  }, [open, method, phase, pairCode, stopPoll]);
   443→   443→
   444→   444→  // QR alur pairing (URL /pair/{code})
   445→   445→  useEffect(() => {
   446→   446→    if (!open || method !== "pair" || !pairUrl) return;
   447→   447→    renderQr(pairUrl);
   448→   448→  }, [open, method, pairUrl, renderQr]);
   449→   449→
   450→   450→  const handleBrowserWallet = useCallback(
   451→   451→    async (opt: WalletOption) => {
   452→   452→      if (!opt.provider) return;
   453→   453→      setConnecting(opt.id);
   454→   454→      setError(null);
   455→   455→      try {
   456→   456→        const address = await connectProvider(opt.provider);
   457→   457→        if (!isValidEvmAddress(address)) throw new Error("invalid address");
   458→   458→        const res = await api.login(address.toLowerCase(), refCode ?? undefined);
   459→   459→        saveSession({ address: res.user.walletAddress, walletName: opt.name, loginAt: Date.now() });
   460→   460→        setUser(res.user);
   461→   461→        onClose();
   462→   462→      } catch (err) {
   463→   463→        const msg =
   464→   464→          err instanceof Error
   465→   465→            ? err.message.includes("user rejected") || err.message.includes("denied")
   466→   466→              ? t("errUserRejected")
   467→   467→              : err.message
   468→   468→            : t("errConnect");
