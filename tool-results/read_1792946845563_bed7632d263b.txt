    51→interface Props {

    63→function newPairingCode(): string {
    64→  let c = "";
    65→  for (let i = 0; i < 6; i++) c += Math.floor(Math.random() * 16).toString(16);
    66→  return c;
    67→}
    68→
    69→/** Alamat EVM acak untuk mode demo (simulasi wallet Binance) */
    70→function randomEvmAddress(): string {
    71→  const hex = "0123456789abcdef";
    72→  let a = "0x";
    73→  for (let i = 0; i < 40; i++) a += hex[Math.floor(Math.random() * 16)];
    74→  return a;
    75→}

    77→export default function WalletConnectModal({ open, onClose, refCode }: Props) {
    78→  const setUser = useGameStore((s) => s.setUser);
    79→  const { t } = useT();
    80→
    81→  // ── metode aktif ──
    82→  const [method, setMethod] = useState<Method>("binance");
    83→
    84→  // ── alur SDK Binance (real pairing) ──
    85→  const [sdkStatus, setSdkStatus] = useState<SdkStatus>("loading");
    86→  const [wcUri, setWcUri] = useState("");
    87→  const [sdkErr, setSdkErr] = useState<string | null>(null);
    88→  const providerRef = useRef<W3WProvider>(null);
    89→  const enablingRef = useRef(false);
    90→  const binanceCancelled = useRef(false);
    91→  const uriTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
    92→
    93→  // ── alur pairing server ──
    94→  const [phase, setPhase] = useState<Phase>("connecting");
    95→  const [pairCode, setPairCode] = useState("");
    96→  const [pairUrl, setPairUrl] = useState("");
    97→  const [secsLeft, setSecsLeft] = useState(PAIR_TTL);
    98→  const [copiedPair, setCopiedPair] = useState(false);
    99→  const [error, setError] = useState<string | null>(null);
   100→
   101→  // ── QR bersama ──
   102→  const [qrData, setQrData] = useState("");
   103→
   104→  // fallback wallet browser (EIP-6963 / injected)
   105→  const [showBrowser, setShowBrowser] = useState(false);
   106→  const [options, setOptions] = useState<WalletOption[]>([]);
   107→  const [connecting, setConnecting] = useState<string | null>(null);
   108→
   109→  const mounted = useRef(true);
   110→  const pollIv = useRef<ReturnType<typeof setInterval> | null>(null);
   111→  const pairCodeRef = useRef("");
   112→
   113→  // Pesan i18n terbaru untuk closure SDK (t tidak stabil antar render)
   114→  const ERR_REJECTED = useRef("");
   115→  const ERR_SDK = useRef("");
   116→  useEffect(() => {
   117→    ERR_REJECTED.current = t("errUserRejected");
   118→    ERR_SDK.current = t("sdkFailed");
   119→  });
   120→
   121→  const stopPoll = useCallback(() => {
   122→    if (pollIv.current) {
   123→      clearInterval(pollIv.current);
   124→      pollIv.current = null;
   125→    }
   126→  }, []);
   127→
   128→  const renderQr = useCallback((payload: string) => {
   129→    QRCode.toDataURL(payload, {
   130→      width: 512,
   131→      margin: 1,
   132→      errorCorrectionLevel: "M",
   133→      color: { dark: "#0B0E11", light: "#FFFFFF" },
   134→    })
   135→      .then((url) => {
   136→        if (mounted.current) setQrData(url);
   137→      })
   138→      .catch(() => {});
   139→  }, []);
   140→
   141→  // ══════════════════════════════════════════
   142→  //  METODE 1 — SDK PARTNER RESMI BINANCE
   143→  // ══════════════════════════════════════════
   144→
   145→  const handleBinanceAccounts = useCallback(
   146→    async (accs: string[]) => {
   147→      const addr = accs?.[0];
   148→      if (!addr || !isValidEvmAddress(addr)) return;
   149→      if (!mounted.current || binanceCancelled.current) return;
   150→      setSdkStatus("approving");
   151→      try {
   152→        const res = await api.login(addr.toLowerCase(), refCode ?? undefined);
   153→        if (!mounted.current) return;
   154→        saveSession({
   155→          address: res.user.walletAddress,
   156→          walletName: "Binance Web3 Wallet",
   157→          loginAt: Date.now(),
   158→        });
   159→        setUser(res.user);
   160→        onClose();
   161→      } catch {
   162→        if (mounted.current) {
   163→          setSdkStatus("failed");
   164→          setSdkErr(t("errLogin"));
   165→        }
   166→      }
   167→    },
   168→    [refCode, setUser, onClose, t]
   169→  );
   170→
   171→  const clearUriTimer = useCallback(() => {
   172→    if (uriTimerRef.current) {
   173→      clearTimeout(uriTimerRef.current);
   174→      uriTimerRef.current = null;
   175→    }
   176→  }, []);
   177→
   178→  const startBinance = useCallback(async () => {
   179→    if (typeof window === "undefined") return;
   180→    binanceCancelled.current = false;
   181→    setSdkStatus("loading");
   182→    setWcUri("");
   183→    setQrData("");
   184→    setSdkErr(null);
   185→
   186→    try {
   187→      const w = window as unknown as {
   188→        binancew3w?: { ethereum?: EIP1193Provider };
   189→        ethereum?: EIP1193Provider & { isBinance?: boolean };
   190→      };
   191→
   192→      // 1) Di dalam dApp browser Binance App → injected provider resmi
   193→      const injected = w.binancew3w?.ethereum ?? (w.ethereum?.isBinance ? w.ethereum : null);
   194→      if (injected) {
   195→        const accs = (await injected.request({ method: "eth_requestAccounts" })) as string[];
   196→        await handleBinanceAccounts(accs);
   197→        return;
   198→      }
   199→
   200→      // 2) SDK partner resmi Binance (QR WalletConnect asli via relay Binance)
   201→      const { getProvider } = await import("@binance/w3w-ethereum-provider");
   202→      if (binanceCancelled.current || !mounted.current) return;
   203→
   204→      if (!providerRef.current) {
   205→        const p: W3WProvider = getProvider({ chainId: 56, showQrCodeModal: false });
   206→        providerRef.current = p;
   207→
   208→        p.on("uri_ready", (uri: string) => {
   209→          if (binanceCancelled.current || !mounted.current) return;
   210→          clearUriTimer();
   211→          setWcUri(uri);
   212→          renderQr(uri);
   213→          setSdkStatus("waiting");
   214→        });
   215→        p.on("accountsChanged", (accs: string[]) => {
   216→          if (!binanceCancelled.current && accs?.length) handleBinanceAccounts(accs);
   217→        });
   218→        p.on("disconnect", () => {
   219→          if (mounted.current && !binanceCancelled.current) {
   220→            enablingRef.current = false;
   221→            setWcUri("");
   222→            setQrData("");
   223→            setSdkStatus("loading");
   224→          }
   225→        });
   226→      }
   227→
   228→      const p = providerRef.current;
   229→
   230→      // Sudah terhubung sebelumnya (singleton SDK) → langsung masuk
   231→      if (p.connected && p.accounts?.length) {
   232→        await handleBinanceAccounts(p.accounts as string[]);
   233→        return;
   234→      }
   235→
   236→      setSdkStatus("waiting");
   237→      clearUriTimer();
   238→      uriTimerRef.current = setTimeout(() => {
   239→        if (!mounted.current || binanceCancelled.current) return;
   240→        if (!wcUri && !providerRef.current?.connected) {
   241→          setSdkStatus("failed");
   242→          setSdkErr(t("sdkFailed"));
   243→        }
   244→      }, SDK_URI_TIMEOUT);
   245→
   246→      // enable() menunggu persetujuan di Binance App (scan QR / deep link).
   247→      // Jika sesi pending dari bukaan modal sebelumnya masih berjalan,
   248→      // jangan panggil enable() dua kali — promise lama tetap memantau.
   249→      if (!enablingRef.current) {
   250→        enablingRef.current = true;
   251→        p.enable()
   252→          .then((accs: string[]) => {
   253→            enablingRef.current = false;
   254→            clearUriTimer();
   255→            if (!binanceCancelled.current) handleBinanceAccounts(accs);
   256→          })
   257→          .catch((err: unknown) => {
   258→            enablingRef.current = false;
   259→            clearUriTimer();
   260→            if (binanceCancelled.current || !mounted.current) return;
   261→            const msg = err instanceof Error ? err.message : "";
   262→            setSdkStatus("failed");
   263→            setSdkErr(/reject|denied/i.test(msg) ? ERR_REJECTED.current : ERR_SDK.current);
   264→          });
   265→      }
   266→    } catch {
   267→      if (!binanceCancelled.current && mounted.current) {
   268→        setSdkStatus("failed");
   269→        setSdkErr(t("sdkFailed"));
   270→      }
   271→    }
   272→  }, [clearUriTimer, handleBinanceAccounts, renderQr, t]);
   273→
   274→  /** Tombol "Open Binance sign-in" — deep link resmi dari @binance/w3w-utils */
   275→  const openBinanceApp = useCallback(async () => {
   276→    if (!wcUri) return;
   277→    try {
   278→      const { openBinanceDeepLink } = await import("@binance/w3w-utils");
   279→      openBinanceDeepLink(wcUri);
   280→    } catch {
   281→      /* noop */
   282→    }
   283→  }, [wcUri]);
   284→
   285→  // ══════════════════════════════════════════
   286→  //  METODE 2 — PAIRING LINK SERVER
   287→  // ══════════════════════════════════════════
   288→
   289→  /** Terapkan sesi login dari hasil approve (tanpa membuat user ganda) */
   290→  const applySession = useCallback(
   291→    (user: PublicUser) => {
   292→      saveSession({
   293→        address: user.walletAddress,
   294→        walletName: "Binance Agent Wallet",
   295→        loginAt: Date.now(),
   296→      });
   297→      setUser(user);
   298→      onClose();
   299→    },
   300→    [setUser, onClose]
   301→  );
   302→
   303→  /** Approve dari perangkat yang sama (tombol Confirm) — lewat API agar sesi sama */
   304→  const approveHere = useCallback(async () => {
   305→    if (!pairCodeRef.current) return;
   306→    setPhase("confirming");
   307→    setError(null);
   308→    try {
   309→      const res = await fetch("/api/pair/approve", {
   310→        method: "POST",
   311→        headers: { "Content-Type": "application/json" },
   312→        body: JSON.stringify({ code: pairCodeRef.current }),
   313→      });
   314→      const d = await res.json();
   315→      if (!res.ok || !d?.ok) {
   316→        if (res.status === 410) {
   317→          setPhase("expired");
   318→        } else {
   319→          setError(d?.error ?? t("errLogin"));
   320→          setPhase("connecting");
   321→        }
   322→        return;
   323→      }
   324→      await new Promise((r) => setTimeout(r, 1200)); // animasi konfirmasi singkat
   325→      applySession(d.user);
   326→    } catch {
   327→      if (mounted.current) {
   328→        setError(t("errLogin"));
   329→        setPhase("connecting");
   330→      }
   331→    }
   332→  }, [applySession, t]);
   333→
   334→  const startPolling = useCallback(() => {
   335→    stopPoll();
   336→    pollIv.current = setInterval(async () => {
   337→      if (!pairCodeRef.current) return;
   338→      try {
   339→        const res = await fetch(`/api/pair/status?code=${pairCodeRef.current}`);
   340→        const d = await res.json();
   341→        if (!mounted.current) return;
   342→        if (d?.status === "APPROVED" && d?.user) {
   343→          stopPoll();
   344→          applySession(d.user);
   345→        } else if (d?.status === "EXPIRED" || d?.status === "NOT_FOUND") {
   346→          stopPoll();
   347→          setPhase("expired");
   348→        }
   349→      } catch {
   350→        /* jaringan sempat gagal → coba di tick berikutnya */
   351→      }
   352→    }, POLL_MS);
   353→  }, [applySession, stopPoll]);
   354→
   355→  const regenPair = useCallback(() => {
   356→    const code = newPairingCode();
   357→    pairCodeRef.current = code;
   358→    setPairCode(code);
   359→    setPairUrl(`${window.location.origin}/pair/${code}`);
   360→    setSecsLeft(PAIR_TTL);
   361→    setQrData("");
   362→    setPhase("connecting");
   363→    setError(null);
   364→    // daftarkan sesi ke server (dengan referral bila ada)
   365→    fetch("/api/pair/create", {
   366→      method: "POST",
   367→      headers: { "Content-Type": "application/json" },
   368→      body: JSON.stringify({ code, refCode: refCode ?? undefined }),
   369→    })
   370→      .then(() => {
   371→        startPolling();
   372→      })
   373→      .catch(() => {
   374→        startPolling(); // tetap polling; approve tanpa create → NOT_FOUND → expired
   375→      });
   376→  }, [refCode, startPolling]);
   377→
   378→  // ── ganti metode ──
   379→  // Ref ke fungsi terbaru (identitas stabil) agar effect buka-modal
   380→  // tidak re-run akibat fungsi baru tiap render (t tidak stabil).
   381→  const latestRef = useRef({ startBinance, regenPair, stopPoll, clearUriTimer });
   382→  useEffect(() => {
   383→    latestRef.current = { startBinance, regenPair, stopPoll, clearUriTimer };
   384→  });
   385→
   386→  const switchMethod = useCallback(
   387→    (m: Method) => {
   388→      if (m === method) return;
   389→      setMethod(m);
   390→      setError(null);
   391→      setQrData("");
   392→      if (m === "pair") {
   393→        binanceCancelled.current = true; // hentikan alur SDK
   394→        latestRef.current.clearUriTimer();
   395→        latestRef.current.regenPair();
   396→      } else {
   397→        latestRef.current.stopPoll();
   398→        latestRef.current.startBinance();
   399→      }
   400→    },
   401→    [method]
   402→  );
   403→
   404→  // Reset tiap kali modal dibuka (HANYA saat `open` berubah)
   405→  useEffect(() => {
   406→    if (!open) return;
   407→    mounted.current = true;
   408→    setShowBrowser(false);
   409→    setOptions([]);
   410→    setCopiedPair(false);
   411→    setMethod("binance");
   412→    latestRef.current.startBinance();
   413→    // deteksi wallet browser secara diam-diam untuk fallback
   414→    getWalletOptions()
   415→      .then((opts) => {
   416→        if (mounted.current) setOptions(opts);
   417→      })
   418→      .catch(() => {});
   419→    return () => {
   420→      mounted.current = false;
   421→      binanceCancelled.current = true;
   422→      latestRef.current.clearUriTimer();
   423→      latestRef.current.stopPoll();
   424→    };
   425→  }, [open]);
   426→
   427→  // Countdown 300s → expired (hanya alur pairing server)
   428→  useEffect(() => {
   429→    if (!open || method !== "pair" || phase !== "connecting") return;
   430→    const iv = setInterval(() => {
   431→      setSecsLeft((s) => {
   432→        if (s <= 1) {
   433→          clearInterval(iv);
   434→          stopPoll();
   435→          if (mounted.current) setPhase("expired");
   436→          return 0;
   437→        }
   438→        return s - 1;
   439→      });
   440→    }, 1000);
   441→    return () => clearInterval(iv);
   442→  }, [open, method, phase, pairCode, stopPoll]);
   443→
   444→  // QR alur pairing (URL /pair/{code})
   445→  useEffect(() => {
   446→    if (!open || method !== "pair" || !pairUrl) return;
   447→    renderQr(pairUrl);
   448→  }, [open, method, pairUrl, renderQr]);
   449→
   450→  const handleBrowserWallet = useCallback(
   451→    async (opt: WalletOption) => {
   452→      if (!opt.provider) return;
   453→      setConnecting(opt.id);
   454→      setError(null);
   455→      try {
   456→        const address = await connectProvider(opt.provider);
   457→        if (!isValidEvmAddress(address)) throw new Error("invalid address");
   458→        const res = await api.login(address.toLowerCase(), refCode ?? undefined);
   459→        saveSession({ address: res.user.walletAddress, walletName: opt.name, loginAt: Date.now() });
   460→        setUser(res.user);
   461→        onClose();
   462→      } catch (err) {
   463→        const msg =
   464→          err instanceof Error
   465→            ? err.message.includes("user rejected") || err.message.includes("denied")
   466→              ? t("errUserRejected")
   467→              : err.message
   468→            : t("errConnect");
   469→        setError(msg);
   470→      } finally {
   471→        if (mounted.current) setConnecting(null);
   472→      }
   473→    },
   474→    [refCode, setUser, onClose, t]
   475→  );
   476→
   477→  const copyPair = async () => {
   478→    try {
   479→      await navigator.clipboard.writeText(pairCode);
   480→      setCopiedPair(true);
   481→      setTimeout(() => setCopiedPair(false), 1500);
   482→    } catch {
   483→      /* ignore */
   484→    }
   485→  };
   486→
   487→  const mmss = `${String(Math.floor(secsLeft / 60)).padStart(2, "0")}:${String(secsLeft % 60).padStart(2, "0")}`;
   488→
   489→  // status badge global
   490→  const busy =
   491→    (method === "binance" && (sdkStatus === "approving" || (sdkStatus === "waiting" && phase === "confirming"))) ||
   492→    (method === "pair" && phase === "confirming");
   493→  const expired = (method === "binance" && sdkStatus === "failed") || (method === "pair" && phase === "expired");
   494→
   495→  return (
   496→    <Dialog open={open} onOpenChange={(o) => !o && onClose()}>
   497→      <DialogContent className="max-w-md rounded-2xl border-[#2B3139] bg-[#181A20] text-[#EAECEF] p-0 gap-0 overflow-hidden">
   498→        <DialogHeader className="flex flex-row items-center justify-between border-b border-[#2B3139] px-5 py-4">
   499→          <DialogTitle className="flex items-center gap-2 text-base font-bold">
   500→            <span className="flex h-8 w-8 items-center justify-center rounded-lg bg-[#F0B90B] text-[#0B0E11]">
   501→              <Wallet className="h-4.5 w-4.5" />
   502→            </span>
   503→            {t("agentWalletTitle")}
   504→          </DialogTitle>
   505→          {/* Status badge */}
   506→          <span
   507→            className={`flex items-center gap-1.5 rounded-full px-2.5 py-1 text-[11px] font-semibold ${
   508→              busy
   509→                ? "bg-[#20B26C]/15 text-[#20B26C]"
   510→                : expired
   511→                  ? "bg-[#F6465D]/15 text-[#F6465D]"
   512→                  : "bg-[#F0B90B]/15 text-[#F0B90B]"
   513→            }`}
   514→          >
   515→            {busy ? (
   516→              <>
   517→                <span className="relative flex h-2 w-2">
   518→                  <span className="absolute inline-flex h-full w-full animate-ping rounded-full bg-[#20B26C] opacity-75" />
   519→                  <span className="relative inline-flex h-2 w-2 rounded-full bg-[#20B26C]" />
   520→                </span>
   521→                {t("connected")}
   522→              </>
   523→            ) : expired ? (
   524→              <>
   525→                <AlertTriangle className="h-3 w-3" />
   526→                {t("expiredBadge")}
   527→              </>
   528→            ) : (
   529→              <>
   530→                <span className="relative flex h-2 w-2">
   531→                  <span className="absolute inline-flex h-full w-full animate-ping rounded-full bg-[#F0B90B] opacity-75" />
   532→                  <span className="relative inline-flex h-2 w-2 rounded-full bg-[#F0B90B]" />
   533→                </span>
   534→                {t("connectStatus")}
   535→              </>
   536→            )}
   537→          </span>
   538→        </DialogHeader>
   539→
   540→        {/* ── Pemilih metode ── */}
   541→        <div className="px-5 pt-4">
   542→          <div className="grid grid-cols-2 gap-1 rounded-xl bg-[#0B0E11] p-1">
   543→            <button
   544→              onClick={() => switchMethod("binance")}
   545→              className={`flex items-center justify-center gap-1.5 rounded-lg py-2 text-xs font-bold transition-colors ${
   546→                method === "binance"
   547→                  ? "bg-[#F0B90B] text-[#0B0E11]"
   548→                  : "text-[#848E9C] hover:text-[#EAECEF]"
   549→              }`}
   550→            >
   551→              <Smartphone className="h-3.5 w-3.5" />
   552→              {t("methodBinance")}
