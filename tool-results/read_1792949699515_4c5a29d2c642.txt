    53→// Metadata dApp resmi yang tampil di layar approve Binance Wallet.
    54→// SDK membaca <meta name="name">/og:site_name (dipasang di layout.tsx), tapi
    55→// kita patch juga clientMeta pada coreConnection agar 100% pasti "Agentic Miner"
    56→// + logo (bukan nama default/wallet aktif user seperti "BLOCXERZ").
    57→function dappMeta() {
    58→  if (typeof window === "undefined") return null;
    59→  return {
    60→    name: "Agentic Miner",
    61→    description: "Mine AGENMINER with AI prompts on BNB Chain",
    62→    url: window.location.origin,
    63→    icons: [`${window.location.origin}/logo.png`],
    64→  };
    65→}
    66→
    67→/** Pastikan clientMeta SDK = Agentic Miner (patch coreConnection.signClient) */
    68→function patchDappMeta(p: unknown) {
    69→  try {
    70→    const meta = dappMeta();
    71→    if (!meta) return;
    72→    const provider = p as { signClient?: { coreConnection?: { clientMeta?: unknown } } };
    73→    const core = provider?.signClient?.coreConnection;
    74→    if (core) core.clientMeta = meta;
    75→  } catch {
    76→    /* best-effort */
    77→  }
    78→}

    92→function newPairingCode(): string {
    93→  let c = "";
    94→  for (let i = 0; i < 6; i++) c += Math.floor(Math.random() * 16).toString(16);
    95→  return c;
    96→}

    99→function randomEvmAddress(): string {
   100→  const hex = "0123456789abcdef";
   101→  let a = "0x";
   102→  for (let i = 0; i < 40; i++) a += hex[Math.floor(Math.random() * 16)];
   103→  return a;
   104→}

   129→
   130→  // ── QR bersama ──

   132→
   133→  // fallback wallet browser (EIP-6963 / injected)
   134→  const [showBrowser, setShowBrowser] = useState(false);

   145→  useEffect(() => {

   147→    ERR_SDK.current = t("sdkFailed");

   152→      clearInterval(pollIv.current);

   154→    }

   159→      width: 512,
   160→      margin: 1,
   161→      errorCorrectionLevel: "M",

   164→      .then((url) => {
   165→        if (mounted.current) setQrData(url);
   166→      })
   167→      .catch(() => {});
   168→  }, []);
   169→
   170→  // ══════════════════════════════════════════
   171→  //  METODE 1 — SDK PARTNER RESMI BINANCE

   173→
   174→  const handleBinanceAccounts = useCallback(
   175→    async (accs: string[]) => {
   176→      const addr = accs?.[0];
   177→      if (!addr || !isValidEvmAddress(addr)) return;
   178→      if (!mounted.current || binanceCancelled.current) return;

   181→        // ── GATE AGENTIC ID: wajib holder NFT SPACE ID Agent ID ──

   183→        if (!mounted.current || binanceCancelled.current) return;
   184→        if (!probe.hasNft) {

   186→          setSdkErr(t("agentIdRequired"));

   212→
   213→  const clearUriTimer = useCallback(() => {
   214→    if (uriTimerRef.current) {
   215→      clearTimeout(uriTimerRef.current);
   216→      uriTimerRef.current = null;

   220→  const startBinance = useCallback(async () => {
   221→    if (typeof window === "undefined") return;
   222→    binanceCancelled.current = false;
   223→    setSdkStatus("loading");
   224→    setWcUri("");
   225→    setQrData("");
   226→    setSdkErr(null);
   227→
   228→    try {
   229→      const w = window as unknown as {
   230→        binancew3w?: { ethereum?: EIP1193Provider };
   231→        ethereum?: EIP1193Provider & { isBinance?: boolean };
   232→      };
   233→
   234→      // 1) Di dalam dApp browser Binance App → injected provider resmi
   235→      const injected = w.binancew3w?.ethereum ?? (w.ethereum?.isBinance ? w.ethereum : null);
   236→      if (injected) {
   237→        const accs = (await injected.request({ method: "eth_requestAccounts" })) as string[];

   239→        return;
   240→      }
   241→
   242→      // 2) SDK partner resmi Binance (QR WalletConnect asli via relay Binance)
   243→      const { getProvider } = await import("@binance/w3w-ethereum-provider");
   244→      if (binanceCancelled.current || !mounted.current) return;
   245→

   254→          clearUriTimer();

   256→          renderQr(uri);
   257→          setSdkStatus("waiting");
   258→        });
   259→        p.on("accountsChanged", (accs: string[]) => {
   260→          if (!binanceCancelled.current && accs?.length) handleBinanceAccounts(accs);
   261→        });
   262→        p.on("disconnect", () => {
   263→          if (mounted.current && !binanceCancelled.current) {
   264→            enablingRef.current = false;
   265→            setWcUri("");
   266→            setQrData("");
   267→            setSdkStatus("loading");
   268→          }
   269→        });

   272→      const p = providerRef.current;
   273→

   278→      }

   280→      setSdkStatus("waiting");
   281→      clearUriTimer();

   283→        if (!mounted.current || binanceCancelled.current) return;
   284→        if (!wcUri && !providerRef.current?.connected) {
   285→          setSdkStatus("failed");
   286→          setSdkErr(t("sdkFailed"));

   292→      patchDappMeta(p);

   299→        p.enable()
   300→          .then((accs: string[]) => {

   306→            enablingRef.current = false;
   307→            clearUriTimer();

   309→            const msg = err instanceof Error ? err.message : "";
   310→            setSdkStatus("failed");
   311→            setSdkErr(/reject|denied/i.test(msg) ? ERR_REJECTED.current : ERR_SDK.current);
   312→          });
   313→      }
   314→    } catch {
   315→      if (!binanceCancelled.current && mounted.current) {
   316→        setSdkStatus("failed");
   317→        setSdkErr(t("sdkFailed"));
   318→      }
   319→    }
   320→  }, [clearUriTimer, handleBinanceAccounts, renderQr, t]);
   321→

   326→      const { openBinanceDeepLink } = await import("@binance/w3w-utils");

   334→  //  METODE 2 — PAIRING LINK SERVER

   337→  /** Terapkan sesi login dari hasil approve (tanpa membuat user ganda) */

   339→    (user: PublicUser) => {
   340→      saveSession({
   341→        address: user.walletAddress,
   342→        walletName: "Binance Agent Wallet",
   343→        loginAt: Date.now(),
   344→      });
   345→      setUser(user);
   346→      onClose();
   347→    },
   348→    [setUser, onClose]

   351→  /** Approve dari perangkat yang sama (tombol Confirm) — lewat API agar sesi sama */
   352→  const approveHere = useCallback(async () => {
   353→    if (!pairCodeRef.current) return;
   354→    setPhase("confirming");
   355→    setError(null);
   356→    try {
   357→      const res = await fetch("/api/pair/approve", {
   358→        method: "POST",
   359→        headers: { "Content-Type": "application/json" },
   360→        body: JSON.stringify({ code: pairCodeRef.current }),
   361→      });
   362→      const d = await res.json();
   363→      if (!res.ok || !d?.ok) {
   364→        if (res.status === 410) {
   365→          setPhase("expired");
   366→        } else {
   367→          setError(d?.error ?? t("errLogin"));
   368→          setPhase("connecting");
   369→        }
   370→        return;
   371→      }
   372→      await new Promise((r) => setTimeout(r, 1200)); // animasi konfirmasi singkat
   373→      applySession(d.user);
   374→    } catch {
   375→      if (mounted.current) {
   376→        setError(t("errLogin"));
   377→        setPhase("connecting");
   378→      }
   379→    }
   380→  }, [applySession, t]);

   386→      try {
   387→        const res = await fetch(`/api/pair/status?code=${pairCodeRef.current}`);
   388→        const d = await res.json();
   389→        if (!mounted.current) return;
   390→        if (d?.status === "APPROVED" && d?.user) {
   391→          stopPoll();
   392→          applySession(d.user);
   393→        } else if (d?.status === "EXPIRED" || d?.status === "NOT_FOUND") {
   394→          stopPoll();
   395→          setPhase("expired");
   396→        }
   397→      } catch {
   398→        /* jaringan sempat gagal → coba di tick berikutnya */
   399→      }
   400→    }, POLL_MS);
   401→  }, [applySession, stopPoll]);
   402→
   403→  const regenPair = useCallback(() => {
   404→    const code = newPairingCode();
   405→    pairCodeRef.current = code;

   407→    setPairUrl(`${window.location.origin}/pair/${code}`);

   414→      method: "POST",
   415→      headers: { "Content-Type": "application/json" },

   419→        startPolling();

   429→  const latestRef = useRef({ startBinance, regenPair, stopPoll, clearUriTimer });

   438→      setError(null);
   439→      setQrData("");

   443→        latestRef.current.regenPair();
   444→      } else {
   445→        latestRef.current.stopPoll();
   446→        latestRef.current.startBinance();
   447→      }
   448→    },
   449→    [method]
   450→  );
   451→
   452→  // Reset tiap kali modal dibuka (HANYA saat `open` berubah)

   454→    if (!open) return;

   456→    setShowBrowser(false);
   457→    setOptions([]);
   458→    setCopiedPair(false);
   459→    setMethod("binance");

   463→      .then((opts) => {
   464→        if (mounted.current) setOptions(opts);
   465→      })
   466→      .catch(() => {});
   467→    return () => {
   468→      mounted.current = false;
   469→      binanceCancelled.current = true;
   470→      latestRef.current.clearUriTimer();
   471→      latestRef.current.stopPoll();
   472→    };
   473→  }, [open]);
   474→
   475→  // Countdown 300s → expired (hanya alur pairing server)
   476→  useEffect(() => {
   477→    if (!open || method !== "pair" || phase !== "connecting") return;
   478→    const iv = setInterval(() => {
   479→      setSecsLeft((s) => {
   480→        if (s <= 1) {
   481→          clearInterval(iv);
   482→          stopPoll();
   483→          if (mounted.current) setPhase("expired");
   484→          return 0;
   485→        }
   486→        return s - 1;
   487→      });
   488→    }, 1000);
   489→    return () => clearInterval(iv);
   490→  }, [open, method, phase, pairCode, stopPoll]);
   491→
   492→  // QR alur pairing (URL /pair/{code})
   493→  useEffect(() => {
   494→    if (!open || method !== "pair" || !pairUrl) return;
   495→    renderQr(pairUrl);
   496→  }, [open, method, pairUrl, renderQr]);
   497→
   498→  const handleBrowserWallet = useCallback(
   499→    async (opt: WalletOption) => {
   500→      if (!opt.provider) return;
   501→      setConnecting(opt.id);
   502→      setError(null);
   503→      try {
   504→        const address = await connectProvider(opt.provider);
   505→        if (!isValidEvmAddress(address)) throw new Error("invalid address");
   506→        // ── GATE AGENTIC ID: wajib holder NFT SPACE ID Agent ID ──
   507→        const probe = await probeAgentId(address.toLowerCase());
   508→        if (mounted.current && !probe.hasNft) {
   509→          throw Object.assign(new Error(t("agentIdRequired")), { gated: true });
   510→        }
   511→        const res = await api.login(address.toLowerCase(), refCode ?? undefined);
   512→        saveSession({ address: res.user.walletAddress, walletName: opt.name, loginAt: Date.now() });
   513→        setUser(res.user);
   514→        onClose();
   515→      } catch (err) {
   516→        const e = err as Error & { gated?: boolean };
   517→        if (mounted.current && e?.gated) {
   518→          setError(e.message);
   519→        } else {
   520→          const msg =
   521→            err instanceof Error
   522→              ? err.message.includes("user rejected") || err.message.includes("denied")
   523→                ? t("errUserRejected")
   524→                : err.message
   525→              : t("errConnect");
   526→          setError(msg);
   527→        }
   528→      } finally {
   529→        if (mounted.current) setConnecting(null);
   530→      }
   531→    },
   532→    [refCode, setUser, onClose, t]
   533→  );
   534→
   535→  const copyPair = async () => {
   536→    try {
   537→      await navigator.clipboard.writeText(pairCode);
   538→      setCopiedPair(true);
   539→      setTimeout(() => setCopiedPair(false), 1500);
   540→    } catch {
   541→      /* ignore */
   542→    }
   543→  };
   544→
   545→  const mmss = `${String(Math.floor(secsLeft / 60)).padStart(2, "0")}:${String(secsLeft % 60).padStart(2, "0")}`;
   546→
   547→  // status badge global
   548→  const busy =
   549→    (method === "binance" && (sdkStatus === "approving" || (sdkStatus === "waiting" && phase === "confirming"))) ||
   550→    (method === "pair" && phase === "confirming");
   551→  const expired = (method === "binance" && sdkStatus === "failed") || (method === "pair" && phase === "expired");
   552→
   553→  return (
